**Step 3 – Edge‑case & exception analysis**

| Situation | Why it matters | Expected behaviour |
|-----------|----------------|--------------------|
| **`def` is `null` / `undefined`** | The constructor forwards the argument to `core.$ZodIntersection.init`. Passing a non‑object should cause a runtime error (most Zod internals validate the definition). | The constructor should throw (or the init routine should throw). |
| **`def` misses required fields (`left` / `right`)** | An intersection without the two operand schemas is invalid. | An error should be thrown during construction. |
| **`def` contains non‑schema values for `left` / `right`** | The intersection must be built from other Zod schemas; giving a plain value is a misuse. | Construction should fail (throw). |
| **Instance methods inherited from `_ZodMiniType` behave correctly** | Even though the methods are not declared here, they are part of the public API and must work with the intersection internals. | `parse` should succeed when both sub‑schemas accept the value and fail otherwise. |
| **`parse` with a value that satisfies *only one* side** | Intersection semantics require the value to satisfy *both* sides. | `parse` must throw a validation error. |
| **`parse` with a value that satisfies *both* sides** | Normal successful path. | `parse` returns the value unchanged. |

import { describe, expect, test } from 'vitest';
import { ZodMiniIntersection } from './schemas.js';

// Helper: a tiny mock Zod schema that only validates a specific literal value
function literalSchema<T extends string | number | boolean>(expected: T) {
  return {
    // the real Zod schema objects expose a `parse` method; we mimic the same signature
    parse(value: unknown) {
      if (value !== expected) {
        throw new Error(`Expected literal ${expected}`);
      }
      return value as T;
    },
    // `safeParse` is also part of the public API of _ZodMiniType
    safeParse(value: unknown) {
      try {
        return { success: true, data: this.parse(value) };
      } catch (e) {
        return { success: false, error: e };
      }
    },
  } as any; // cast to any to satisfy the type system for the test
}

// Minimal valid definition that uses the two mock schemas above
function makeValidDef<A, B>(left: any, right: any) {
  return {
    left,
    right,
    typeName: 'ZodIntersection' as const,
  } as unknown as core.$ZodIntersectionInternals<A, B>;
}

describe('ZodMiniIntersection – constructor edge cases', () => {
  test('throws when definition is undefined', () => {
    const Intersection = ZodMiniIntersection;
    // @ts-expect-error – intentionally passing wrong type
    expect(() => new Intersection(undefined)).toThrowError();
  });

  test('throws when definition is null', () => {
    const Intersection = ZodMiniIntersection;
    // @ts-expect-error – intentionally passing wrong type
    expect(() => new Intersection(null)).toThrowError();
  });

  test('throws when required fields are missing', () => {
    const badDef = {} as any;
    const Intersection = ZodMiniIntersection;
    expect(() => new Intersection(badDef)).toThrowError();
  });

  test('throws when left/right are not Zod‑like objects', () => {
    const badDef = { left: 123, right: 'abc', typeName: 'ZodIntersection' } as any;
    const Intersection = ZodMiniIntersection;
    expect(() => new Intersection(badDef)).toThrowError();
  });
});

describe('ZodMiniIntersection – inherited parsing behaviour', () => {
  // two literal schemas: one expects the number 42, the other expects the string "ok"
  const leftSchema = literalSchema(42);
  const rightSchema = literalSchema('ok');

  const def = makeValidDef(leftSchema, rightSchema);
  const Intersection = ZodMiniIntersection;
  const schema = new Intersection(def);

  test('parse fails when value satisfies only the left side', () => {
    expect(() => (schema as any).parse(42)).toThrowError();
  });

  test('parse fails when value satisfies only the right side', () => {
    expect(() => (schema as any).parse('ok')).toThrowError();
  });

  test('parse succeeds only when value satisfies both sides', () => {
    // we need a value that is *both* 42 and "ok" – impossible for primitives.
    // To make the test realistic we replace the mock schemas with ones that
    // accept the same value.
    const left = literalSchema('shared');
    const right = literalSchema('shared');
    const sharedDef = makeValidDef(left, right);
    const sharedSchema = new Intersection(sharedDef);

    expect((sharedSchema as any).parse('shared')).toBe('shared');
  });

  test('safeParse returns success flag correctly', () => {
    const left = literalSchema('x');
    const right = literalSchema('x');
    const def = makeValidDef(left, right);
    const s = new Intersection(def);

    const ok = (s as any).safeParse('x');
    expect(ok.success).toBe(true);
    expect(ok.data).toBe('x');

    const fail = (s as any).safeParse('y');
    expect(fail.success).toBe(false);
    expect(fail.error).toBeInstanceOf(Error);
  });
});
